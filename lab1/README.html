// 03 CONFIG.YAML STEP 2-3
yaml~/agentgateway-config.yaml
cat > ~/agentgateway-config.yaml << 'YAML'
# yaml-language-server: $schema=https://agentgateway.dev/schema/config
binds:
  - port: 3000
    listeners:
      - routes:
          - policies:             # ← route-level policies
              cors:
                allowOrigins: ["*"]
                allowHeaders: ["*"]
              localRateLimit:
                - maxTokens: 10
                  tokensPerFill: 10
                  fillInterval: 60s
            backends:
              - ai:
                  name: gemini
                  provider:
                    gemini:
                      model: gemini-2.5-flash-lite
                  policies:          # ← backend-level policies
                    backendAuth:
                      key: "$GEMINI_API_KEY"
YAML
STRUCTURE
├─ route-level  : cors, localRateLimit, jwtAuth, basicAuth, apiKey
└─ backend-level: backendAuth, timeout, retry
